Note on the two /pcie nodes: their #address-cells is 3, so the first cell of each child
address is the PCI phys.hi space/flags tag, not part of the numeric address. With the tag
separated out, both windows map 1:1 (0x40100000->0x40100000, 0x40200000->0x40200000 and the
0x6010_0000/0x6020_0000 pair), so they are IDENTITY, not translating. Treating phys.hi as
address bits would misclassify them as translating; this script does not.

4. DESCENDANTS OF TRANSLATING NODES THAT CARRY `reg`
====================================================================================================

For each translating node, every descendant with a `reg` property. `reg` is decoded with
the DESCENDANT'S PARENT'S cell counts, then walked up through each ancestor's `ranges`.

Two kinds of `reg` appear below and they must not be conflated:
  MMIO      -- a bus offset that resolves to a CPU-physical address through the chain.
  GRAPH     -- a `reg` on a port@N / endpoint@N node under a `ports` container. `ports`
               declares no `ranges`, so the value is a port index, not an address at all;
               it is unmappable by construction and is NOT a bus offset for MMIO purposes.

#!/usr/bin/env python3
"""Extract address-translation facts from a flattened device tree blob.

Usage: python3 addr.py <file.dtb>

Walks the DTB, finds every node with a `ranges` property, classifies each as
identity / translating / empty, lists every `simple-bus` node, and for each
translating node resolves the CPU-physical address of every descendant `reg`.
Stdlib only; no external dtc.
"""
import hashlib
import struct
import sys

FDT_BEGIN_NODE, FDT_END_NODE, FDT_PROP, FDT_NOP, FDT_END = 1, 2, 3, 4, 9


class Node:
    def __init__(self, name, parent):
        self.name = name
        self.parent = parent
        self.children = []
        self.props = {}
        if parent is None:
            self.path = "/"
        elif parent.path == "/":
            self.path = "/" + name
        else:
            self.path = parent.path + "/" + name

    def u32(self, prop, default=None):
        d = self.props.get(prop)
        if d is None or len(d) != 4:
            return default
        return struct.unpack(">I", d)[0]

    # Cell counts this node declares for its *children*.
    def ac(self):
        return self.u32("#address-cells", 2)   # DT spec default

    def sc(self):
        return self.u32("#size-cells", 1)      # DT spec default

    def ac_declared(self):
        return self.u32("#address-cells") is not None

    def sc_declared(self):
        return self.u32("#size-cells") is not None

    def compatible(self):
        d = self.props.get("compatible")
        if d is None:
            return []
        return [s.decode(errors="replace") for s in d.rstrip(b"\x00").split(b"\x00") if s]


def parse(blob):
    magic, _ts, off_struct, off_strings = struct.unpack(">4I", blob[:16])
    if magic != 0xD00DFEED:
        raise SystemExit("bad magic 0x%x" % magic)
    p = off_struct
    root = None
    cur = None
    while True:
        tok = struct.unpack(">I", blob[p:p + 4])[0]
        p += 4
        if tok == FDT_BEGIN_NODE:
            end = blob.index(b"\x00", p)
            name = blob[p:end].decode(errors="replace")
            p = (end + 1 + 3) & ~3
            n = Node(name, cur)
            if root is None:
                root = n
            else:
                cur.children.append(n)
            cur = n
        elif tok == FDT_END_NODE:
            cur = cur.parent
        elif tok == FDT_PROP:
            ln, noff = struct.unpack(">II", blob[p:p + 8])
            p += 8
            data = blob[p:p + ln]
            p = (p + ln + 3) & ~3
            end = blob.index(b"\x00", off_strings + noff)
            cur.props[blob[off_strings + noff:end].decode(errors="replace")] = data
        elif tok == FDT_NOP:
            continue
        elif tok == FDT_END:
            break
        else:
            raise SystemExit("bad token %d" % tok)
    return root


def cells(data, off, n):
    """Read n big-endian 32-bit cells starting at cell index off."""
    v = 0
    for i in range(n):
        v = (v << 32) | struct.unpack(">I", data[(off + i) * 4:(off + i + 1) * 4])[0]
    return v


def walk(n, out=None):
    out = [] if out is None else out
    out.append(n)
    for c in n.children:
        walk(c, out)
    return out


def decode_ranges(node):
    """Return (kind, entries). entries = [(child_addr, parent_addr, size)]."""
    d = node.props.get("ranges")
    if d is None:
        return None, []
    if len(d) == 0:
        return "empty", []
    nac, nsc = node.ac(), node.sc()
    pac = node.parent.ac() if node.parent else 2
    step = nac + pac + nsc
    if step == 0 or len(d) % (step * 4) != 0:
        return "unparsable", []
    ents = []
    for i in range(len(d) // (step * 4)):
        b = i * step
        ca = cells(d, b, nac)
        tag = None
        if nac == 3:
            # PCI: the high cell is a phys.hi space/flags tag, not an address.
            tag = ca >> 64
            ca &= (1 << 64) - 1
        ents.append((ca, cells(d, b + nac, pac), cells(d, b + nac + pac, nsc), tag))
    kind = "identity" if all(c == pa for c, pa, _, _ in ents) else "translating"
    return kind, ents


def xlate_one(addr, node):
    """Map one address from node's child address space to node's parent space."""
    kind, ents = decode_ranges(node)
    if kind is None:
        return None, "parent %s has no ranges (non-translatable)" % node.path
    if kind == "empty":
        return addr, "identity (empty ranges)"
    for c, pa, sz, _tag in ents:
        if c <= addr < c + sz:
            return addr - c + pa, "0x%x -> 0x%x via [0x%x,0x%x)+0x%x" % (
                addr, addr - c + pa, c, c + sz, pa)
    return None, "0x%x outside all ranges of %s" % (addr, node.path)


def to_cpu(node, addr):
    """Translate a reg address of `node` (in its parent's space) up to the root."""
    steps = []
    cur = node.parent
    a = addr
    while cur is not None and cur.parent is not None:
        a2, why = xlate_one(a, cur)
        steps.append((cur.path, why))
        if a2 is None:
            return None, steps
        a = a2
        cur = cur.parent
    return a, steps


def regs(node):
    """Return [(addr, size)] for node's reg, decoded with parent's cell counts."""
    d = node.props.get("reg")
    if d is None or node.parent is None:
        return []
    ac, sc = node.parent.ac(), node.parent.sc()
    step = ac + sc
    if step == 0 or len(d) % (step * 4) != 0:
        return []
    return [(cells(d, i * step, ac), cells(d, i * step + ac, sc) if sc else 0)
            for i in range(len(d) // (step * 4))]


def rawhex(node, prop):
    d = node.props.get(prop, b"")
    return "<" + " ".join("0x%08x" % w for w in
                          struct.unpack(">%dI" % (len(d) // 4), d)) + ">" if d else "<>"


def main():
    path = sys.argv[1]
    blob = open(path, "rb").read()
    print("sha256 %s" % hashlib.sha256(blob).hexdigest())
    root = parse(blob)
    nodes = walk(root)
    print("nodes %d" % len(nodes))
    print("root #address-cells=%d #size-cells=%d (declared: %s/%s)" % (
        root.ac(), root.sc(), root.ac_declared(), root.sc_declared()))

    ranged = [n for n in nodes if "ranges" in n.props]
    print("\n=== NODES WITH ranges (%d) ===" % len(ranged))
    for n in ranged:
        kind, ents = decode_ranges(n)
        p = n.parent
        print("%s|%s|self ac=%d sc=%d|parent %s ac=%d sc=%d|compat=%s|raw=%s" % (
            n.path, kind, n.ac(), n.sc(),
            p.path if p else "(none)", p.ac() if p else -1, p.sc() if p else -1,
            ",".join(n.compatible()), rawhex(n, "ranges")))
        for c, pa, sz, tag in ents:
            print("    child=0x%x%s parent=0x%x size=0x%x delta=%+d" % (
                c, "" if tag is None else " (phys.hi=0x%08x)" % tag, pa, sz, pa - c))

    sb = [n for n in nodes if "simple-bus" in n.compatible()]
    print("\n=== simple-bus NODES (%d) ===" % len(sb))
    for n in sb:
        kind, _ = decode_ranges(n)
        print("%s|%s|compat=%s" % (n.path, kind if kind else "NO-ranges",
                                   ",".join(n.compatible())))

    xl = [n for n in ranged if decode_ranges(n)[0] == "translating"]
    print("\n=== DESCENDANTS OF TRANSLATING NODES WITH reg ===")
    total = 0
    gmm = [0]
    for n in xl:
        kids = [k for k in walk(n)[1:] if "reg" in k.props]
        total += len(kids)
        mm = sum(1 for k in kids if all(to_cpu(k, a)[0] is not None for a, _ in regs(k)) and regs(k))
        gmm[0] += mm
        print("-- %s : %d descendant(s) with reg (%d translatable MMIO, %d non-translatable"
              " graph/index nodes)" % (n.path, len(kids), mm, len(kids) - mm))
        for k in kids:
            rs = regs(k)
            outs = []
            for a, s in rs:
                cpu, steps = to_cpu(k, a)
                outs.append("reg=0x%x sz=0x%x -> cpu=%s [%s]" % (
                    a, s, ("0x%x" % cpu) if cpu is not None else "UNMAPPED",
                    "; ".join(w for _, w in steps)))
            print("   %s | raw reg=%s | %s" % (k.path, rawhex(k, "reg"), " ; ".join(outs)))

    print("\n=== TOTALS ===")
    print("simple-bus nodes: %d" % len(sb))
    print("nodes with ranges: %d" % len(ranged))
    print("translating (any node): %d" % len(xl))
    print("translating simple-bus: %d" % len([n for n in sb if decode_ranges(n)[0] == "translating"]))
    print("descendants-with-reg under translating nodes: %d" % total)
    print("  of which translatable MMIO: %d" % gmm[0])
    print("  of which non-translatable (ports/endpoint graph indices): %d" % (total - gmm[0]))

    print("\n=== SPOTLIGHT NODES ===")
    want = []
    for n in nodes:
        c = n.compatible()
        if ("google,level-gia" in c or "arm,smmu-v3" in c or n.name == "reserved-memory"
                or "arm,gic-v3" in c):
            want.append(n)
        elif "reg" in n.props:
            for a, _ in regs(n):
                if a in (0x0DB62000, 0x05880000):
                    want.append(n)
                    break
    for n in want:
        anc = []
        cur = n.parent
        while cur is not None:
            k, _ = decode_ranges(cur)
            anc.append("%s(%s)" % (cur.path, k if k else "no-ranges"))
            cur = cur.parent
        for a, s in regs(n):
            cpu, _ = to_cpu(n, a)
            print("%s | compat=%s | reg=0x%x sz=0x%x | cpu=%s | parent-chain=%s" % (
                n.path, ",".join(n.compatible()), a, s,
                ("0x%x" % cpu) if cpu is not None else "UNMAPPED", " <- ".join(anc)))
        if not regs(n):
            print("%s | compat=%s | (no reg) | parent-chain=%s" % (
                n.path, ",".join(n.compatible()), " <- ".join(anc)))


main()
